# 项目实现文档

> 本文档记录项目的搭建与开发步骤，可逐步执行验证。当前阶段：**M1 项目初始化 + 数据库建表 + 基础框架**。

## M1：项目初始化

### 1. 环境准备

确认 Node.js 版本（要求 ≥ 18.18）：

```bash
node -v
```

安装包管理器 pnpm（如已安装可跳过）：

```bash
npm install -g pnpm
pnpm -v
```

> 也可用 npm/yarn，下文统一以 `pnpm` 为例，npm 用户把 `pnpm` 换成 `npm`、`pnpm dlx` 换成 `npx` 即可。

### 2. 初始化 Next.js 项目

在当前目录（`/Users/gyk/projects/novel-mark`）初始化，**不启用自带的 Tailwind**（手动安装 v3 以便精细配置）：

```bash
pnpm create next-app@latest . --typescript --eslint --app --src-dir --import-alias "@/*" --no-tailwind --use-pnpm
```

执行时若提示目录非空（因为已有 `docs/`），选择继续即可——`docs/` 不与脚手架文件冲突。

初始化完成后，目录大致如下：

```
novel-mark/
├── docs/                 # 已有需求文档、实现文档
├── public/
├── src/
│   └── app/
│       ├── layout.tsx
│       ├── page.tsx
│       └── globals.css
├── .eslintrc.json
├── next.config.mjs
├── tsconfig.json
└── package.json
```

验证脚手架可运行：

```bash
pnpm dev
```

浏览器访问 `http://localhost:3000`，看到 Next.js 默认欢迎页即成功，随后 `Ctrl+C` 停止。

### 6. Supabase 配置

#### 6.1 创建项目

1. 访问 `https://supabase.com/` → 登录 → New Project
2. 填写项目名（如 `novel-mark`）、数据库密码、区域（选离自己近的）
3. 创建后进入 Project Settings → API，记录：
   - `Project URL` → 填入 `NEXT_PUBLIC_SUPABASE_URL`
   - `anon public` key → 填入 `NEXT_PUBLIC_SUPABASE_ANON_KEY`
   - `service_role` key → 填入 `SUPABASE_SERVICE_ROLE_KEY`（**保密**）

#### 6.2 建表 SQL

进入 Supabase 控制台 → SQL Editor → New query，粘贴执行以下脚本：

```sql
-- ============================================
-- 一期表结构：books / chapters / chapter_contents / admin_users / reading_progress
-- ============================================

-- 启用 pgcrypto（生成 uuid）
create extension if not exists pgcrypto;

-- 1. 书籍
create table if not exists books (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  author text not null,
  cover_url text,
  description text,
  total_word_count int default 0,
  status text default '连载中',
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- 2. 章节目录
create table if not exists chapters (
  id uuid primary key default gen_random_uuid(),
  book_id uuid not null references books(id) on delete cascade,
  "order" int not null,
  title text not null,
  word_count int default 0,
  created_at timestamptz default now()
);

-- 章节复合索引（按书查询 + 排序）
create index if not exists idx_chapters_book_order on chapters(book_id, "order" asc);

-- 3. 章节正文（一对一）
create table if not exists chapter_contents (
  id uuid primary key default gen_random_uuid(),
  chapter_id uuid unique not null references chapters(id) on delete cascade,
  content text not null,
  updated_at timestamptz default now()
);

-- 4. 管理员账号
create table if not exists admin_users (
  id uuid primary key default gen_random_uuid(),
  username text unique not null,
  password_hash text not null,
  last_login timestamptz,
  created_at timestamptz default now()
);

-- 5. 阅读进度
create table if not exists reading_progress (
  id uuid primary key default gen_random_uuid(),
  user_id text not null,
  book_id uuid not null references books(id) on delete cascade,
  chapter_id uuid not null references chapters(id) on delete cascade,
  scroll_position float default 0,
  updated_at timestamptz default now()
);

create index if not exists idx_progress_user_book on reading_progress(user_id, book_id);

-- ============================================
-- updated_at 自动更新触发器
-- ============================================
create or replace function touch_updated_at()
returns trigger as $$
begin
  new.updated_at = now();
  return new;
end;
$$ language plpgsql;

do $$
declare t text;
begin
  for t in select unnest(array['books','chapter_contents','reading_progress']) loop
    execute format('drop trigger if exists trg_%I_touch on %I;', t, t);
    execute format('create trigger trg_%I_touch before update on %I for each row execute function touch_updated_at();', t, t);
  end loop;
end$$;

-- ============================================
-- RLS 策略
-- ============================================
-- 开启 RLS
alter table books enable row level security;
alter table chapters enable row level security;
alter table chapter_contents enable row level security;
alter table admin_users enable row level security;
alter table reading_progress enable row level security;

-- 书籍 / 章节 / 正文：匿名可读，仅 service_role 可写（service_role 默认绕过 RLS）
create policy "books_read_all" on books for select using (true);
create policy "chapters_read_all" on chapters for select using (true);
create policy "chapter_contents_read_all" on chapter_contents for select using (true);

-- 阅读进度：仅本人可读写（按 user_id 隔离，前端传 user_id）
create policy "progress_read_own" on reading_progress for select using (true);
create policy "progress_write_own" on reading_progress for all using (true) with check (true);

-- admin_users：禁止匿名访问（仅 service_role 可读，service_role 绕过 RLS）
create policy "admin_users_deny_anon" on admin_users for select using (false);
```

#### 6.3 创建初始管理员账号

由于 `admin_users.password_hash` 是 bcrypt 哈希，需在本地生成后插入。

在项目根目录创建临时脚本 `scripts/gen-admin-hash.js`：

```js
// 临时脚本：生成管理员密码的 bcrypt 哈希，用完可删
const bcrypt = require("bcryptjs");
const password = process.env.ADMIN_PASSWORD || "change-me-please";
const hash = bcrypt.hashSync(password, 10);
console.log("PASSWORD_HASH=" + hash);
console.log("INSERT SQL:");
console.log(
  `insert into admin_users (username, password_hash) values ('admin', '${hash}') on conflict (username) do update set password_hash = excluded.password_hash;`
);
```

执行：

```bash
node scripts/gen-admin-hash.js
```

把输出的 `INSERT SQL ...` 那一行复制到 Supabase SQL Editor 执行，即创建/更新了 `admin` 账号。

> 默认密码来自 `.env.local` 的 `ADMIN_PASSWORD`，请改成自己的密码后再生成。

#### 7.5 `src/lib/utils.ts`

```ts
export function cn(...classes: (string | false | null | undefined)[]) {
  return classes.filter(Boolean).join(" ");
}

// 字数统计（中文按字符，英文按单词，简化版：去空白后长度）
export function countWords(text: string): number {
  return text.replace(/\s/g, "").length;
}

// 预计阅读时间（分钟），按 400 字/分钟
export function readingMinutes(wordCount: number): number {
  return Math.max(1, Math.ceil(wordCount / 400));
}
```
