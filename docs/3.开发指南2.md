### 2.3 编写数据查询封装

#### `src/lib/supabase/queries.ts`

封装所有前台会用到的查询，避免在组件里写裸 SQL：

```ts
import { supabaseServer } from "./server";
import type { Book, Chapter, ChapterContent, Volume, VolumeWithChapters } from "@/types/database";

export async function fetchBooks(): Promise<Book[]> {
  const supabase = supabaseServer();
  const { data, error } = await supabase
    .from("books")
    .select("*")
    .order("created_at", { ascending: false });
  if (error) throw error;
  return (data as Book[]) || [];
}

export async function fetchBook(bookId: string): Promise<Book | null> {
  const supabase = supabaseServer();
  const { data, error } = await supabase
    .from("books")
    .select("*")
    .eq("id", bookId)
    .single();
  if (error) throw error;
  return (data as Book) || null;
}

export async function fetchVolumes(bookId: string): Promise<Volume[]> {
  const supabase = supabaseServer();
  const { data, error } = await supabase
    .from("volumes")
    .select("*")
    .eq("book_id", bookId)
    .order("order", { ascending: true });
  if (error) throw error;
  return (data as Volume[]) || [];
}

export async function fetchChapters(bookId: string): Promise<Chapter[]> {
  const supabase = supabaseServer();
  const { data, error } = await supabase
    .from("chapters")
    .select("*")
    .eq("book_id", bookId)
    .order("order", { ascending: true });
  if (error) throw error;
  return (data as Chapter[]) || [];
}

/** 获取按卷分组的目录数据 */
export async function fetchVolumeWithChapters(bookId: string): Promise<VolumeWithChapters[]> {
  const [volumes, chapters] = await Promise.all([
    fetchVolumes(bookId),
    fetchChapters(bookId),
  ]);

  // 如果没有卷，创建一个默认"正文"卷包含所有章节
  if (volumes.length === 0) {
    return [{
      id: "",
      book_id: bookId,
      title: "正文",
      order: 0,
      created_at: "",
      chapters,
    }];
  }

  // 按 volume_id 分组
  const volumeMap = new Map<string, VolumeWithChapters>();
  for (const v of volumes) {
    volumeMap.set(v.id, { ...v, chapters: [] });
  }

  for (const chapter of chapters) {
    const vid = chapter.volume_id;
    if (vid && volumeMap.has(vid)) {
      volumeMap.get(vid)!.chapters.push(chapter);
    } else {
      // 没有卷归属的章节，放入第一个卷
      const firstKey = volumes[0]?.id;
      if (firstKey) {
        volumeMap.get(firstKey)!.chapters.push(chapter);
      }
    }
  }

  return volumes.map((v) => volumeMap.get(v.id)!);
}

export async function fetchChapterWithContent(
  chapterId: string
): Promise<{ chapter: Chapter; content: ChapterContent } | null> {
  const supabase = supabaseServer();
  const { data, error } = await supabase
    .from("chapters")
    .select(`
      *,
      chapter_contents (*)
    `)
    .eq("id", chapterId)
    .single();
  if (error) throw error;
  if (!data) return null;
  return {
    chapter: data as Chapter,
    content: (data as any).chapter_contents?.[0] || null,
  };
}
```

#### 新建 `src/app/(site)/book/[bookId]/[chapterId]/page.tsx`

```tsx
import { notFound, redirect } from "next/navigation";
import { fetchBook, fetchChapters, fetchChapterWithContent } from "@/lib/supabase/queries";
import { Reader } from "@/components/site/Reader";
import { readingMinutes } from "@/lib/utils";

export default async function ChapterPage({
  params,
}: {
  params: Promise<{ bookId: string; chapterId: string }>;
}) {
  const { bookId, chapterId } = await params;

  const [book, chapters, result] = await Promise.all([
    fetchBook(bookId),
    fetchChapters(bookId),
    fetchChapterWithContent(chapterId),
  ]);

  if (!book) notFound();
  if (!result) notFound();

  // 计算上一章 / 下一章
  const currentIndex = chapters.findIndex((c) => c.id === chapterId);
  const prevChapter = currentIndex > 0 ? chapters[currentIndex - 1] : null;
  const nextChapter =
    currentIndex < chapters.length - 1 ? chapters[currentIndex + 1] : null;

  return (
    <Reader
      bookId={book.id}
      bookTitle={book.title}
      chapter={result.chapter}
      content={result.content?.content ?? ""}
      prevChapter={prevChapter}
      nextChapter={nextChapter}
      readingMinutes={readingMinutes(result.chapter.word_count)}
    />
  );
}
```

#### 新建 `src/components/site/Reader.tsx`

```tsx
"use client";

import { useEffect, useRef, useState, useCallback } from "react";
import Link from "next/link";
import type { Chapter } from "@/types/database";

type Props = {
  bookId: string;
  bookTitle: string;
  chapter: Chapter;
  content: string;
  prevChapter: Chapter | null;
  nextChapter: Chapter | null;
  readingMinutes: number;
};

export function Reader({
  bookId,
  bookTitle,
  chapter,
  content,
  prevChapter,
  nextChapter,
  readingMinutes,
}: Props) {
  const [progress, setProgress] = useState(0);
  const contentRef = useRef<HTMLDivElement>(null);

  // 阅读进度：计算内容区域相对于视口的滚动进度
  useEffect(() => {
    function onScroll() {
      const el = contentRef.current;
      if (!el) return;
      const rect = el.getBoundingClientRect();
      const viewportH = window.innerHeight;
      const total = rect.height - viewportH;
      if (total <= 0) {
        setProgress(100);
        return;
      }
      const scrolled = Math.min(Math.max(-rect.top, 0), total);
      setProgress(Math.round((scrolled / total) * 100));
    }
    window.addEventListener("scroll", onScroll, { passive: true });
    onScroll();
    return () => window.removeEventListener("scroll", onScroll);
  }, [content]);

  // 将正文按段落切分
  const paragraphs = content
    .split(/\n+/)
    .map((p) => p.trim())
    .filter(Boolean);

  return (
    <>
      {/* 顶部阅读进度黄线 */}
      <div className="fixed left-0 top-0 z-50 h-[3px] w-full bg-[var(--border)]">
        <div
          className="h-full bg-[var(--accent)] transition-[width] duration-150 ease-out"
          style={{ width: `${progress}%` }}
        />
      </div>

      <article className="mx-auto max-w-2xl">
        {/* 导航 */}
        <div className="mb-6 flex items-center justify-between text-xs text-[var(--fg-muted)]">
          <Link href={`/book/${bookId}`} className="hover:text-[var(--accent)]">
            ← 返回目录
          </Link>
          <span>{bookTitle}</span>
        </div>

        {/* 章节标题 + 黄短线分隔 */}
        <header className="mb-6">
          <h1 className="font-serif text-2xl mb-3">{chapter.title}</h1>
          <div className="h-[2px] w-8 bg-[var(--accent)] mb-3" />
          <div className="text-xs text-[var(--fg-muted)]">
            <span>{chapter.word_count.toLocaleString()} 字</span>
            <span className="mx-2">·</span>
            <span>约 {readingMinutes} 分钟</span>
          </div>
        </header>

        {/* 正文：带渐显动画 */}
        <div ref={contentRef} className="reading-content text-[var(--fg)]">
          {paragraphs.map((p, i) => (
            <FadeInParagraph key={i} index={i}>
              {p}
            </FadeInParagraph>
          ))}
        </div>

        {/* 翻页 */}
        <nav className="mt-12 flex items-center justify-between border-t border-[var(--border)] pt-8 pb-16">
          {prevChapter ? (
            <Link
              href={`/book/${bookId}/${prevChapter.id}`}
              className="rounded-md border border-[var(--border)] px-4 py-2 text-sm hover:border-[var(--accent)] transition-colors"
            >
              ← {prevChapter.title}
            </Link>
          ) : (
            <span className="text-sm text-[var(--fg-muted)]">已是第一章</span>
          )}
          {nextChapter ? (
            <Link
              href={`/book/${bookId}/${nextChapter.id}`}
              className="rounded-md bg-[var(--accent)] px-4 py-2 text-sm font-medium text-black hover:opacity-90 transition-opacity"
            >
              {nextChapter.title} →
            </Link>
          ) : (
            <span className="text-sm text-[var(--fg-muted)]">已是最后一章</span>
          )}
        </nav>
      </article>
    </>
  );
}

// 段落渐显组件
function FadeInParagraph({
  children,
  index,
}: {
  children: React.ReactNode;
  index: number;
}) {
  const [visible, setVisible] = useState(false);
  const ref = useRef<HTMLParagraphElement>(null);

  const setRef = useCallback(
    (node: HTMLParagraphElement | null) => {
      ref.current = node;
      if (!node) return;

      // 前 3 段立即显示
      if (index < 3) {
        setVisible(true);
        return;
      }

      const observer = new IntersectionObserver(
        (entries) => {
          if (entries[0].isIntersecting) {
            setVisible(true);
            observer.disconnect();
          }
        },
        { threshold: 0.1, rootMargin: "0px 0px -50px 0px" }
      );
      observer.observe(node);
    },
    [index]
  );

  return (
    <p
      ref={setRef}
      className={`transition-all duration-700 ease-out ${
        visible
          ? "opacity-100 translate-y-0"
          : "opacity-0 translate-y-4"
      }`}
      style={{ transitionDelay: `${Math.min(index * 50, 200)}ms` }}
    >
      {children}
    </p>
  );
}
```

### 3.3 编写认证工具

#### `src/lib/auth.ts`

```ts
import jwt from "jsonwebtoken";
import { cookies } from "next/headers";

const JWT_SECRET = process.env.JWT_SECRET || "fallback-secret-change-me";
const TOKEN_NAME = "admin_token";

type TokenPayload = {
  adminId: string;
  username: string;
};

export function signToken(payload: TokenPayload): string {
  return jwt.sign(payload, JWT_SECRET, { expiresIn: "7d" });
}

export function verifyToken(token: string): TokenPayload | null {
  try {
    return jwt.verify(token, JWT_SECRET) as TokenPayload;
  } catch {
    return null;
  }
}

export async function getSession(): Promise<TokenPayload | null> {
  const cookieStore = await cookies();
  const token = cookieStore.get(TOKEN_NAME)?.value;
  if (!token) return null;
  return verifyToken(token);
```

```ts
import { NextResponse } from "next/server";
import bcrypt from "bcryptjs";
import { adminLogin } from "@/lib/supabase/admin-queries";
import { signToken, TOKEN_KEY } from "@/lib/auth";

export async function POST(request: Request) {
  const { username, password } = await request.json();

  if (!username || !password) {
    return NextResponse.json({ error: "缺少用户名或密码" }, { status: 400 });
  }

  // 查用户
  const { supabaseAdmin } = await import("@/lib/supabase/admin");
  const { data, error } = await supabaseAdmin()
    .from("admin_users")
    .select("*")
    .eq("username", username)
    .single();

  if (error || !data) {
    return NextResponse.json({ error: "用户名或密码错误" }, { status: 401 });
  }

  // 验密码
  const valid = await bcrypt.compare(password, data.password_hash);
  if (!valid) {
    return NextResponse.json({ error: "用户名或密码错误" }, { status: 401 });
  }

  // 签发 token
  const token = signToken({
    adminId: data.id,
    username: data.username,
  });

  // 更新 last_login
  await supabaseAdmin()
    .from("admin_users")
    .update({ last_login: new Date().toISOString() })
    .eq("id", data.id);

  const response = NextResponse.json({ ok: true, username: data.username });
  response.cookies.set(TOKEN_KEY(), token, {
    httpOnly: true,
    maxAge: 60 * 60 * 24 * 7,
    path: "/",
  });
  return response;
}
```

```ts
export async function adminCreateChapter(
  bookId: string,
  chapter: { title: string; content: string }
): Promise<Chapter> {

  const { data: maxOrderRow } = await supabase
    .from("chapters")

    .eq("book_id", bookId)

    .from("chapters")

    .select()
    .single();

    .insert([{ chapter_id: data.id, content: chapter.content }]);

    .from("chapter_contents")
    .update({ content, updated_at: new Date().toISOString() })
    .eq("chapter_id", chapterId);

  await refreshBookWordCount(bookId);
}

export async function adminDeleteChapter(chapterId: string, bookId: string): Promise<void> {
  const supabase = supabaseAdmin();
  await supabase.from("chapters").delete().eq("id", chapterId);
  // 重新排序剩余章节
  await reorderChapters(bookId);
  await refreshBookWordCount(bookId);
}

async function reorderChapters(bookId: string): Promise<void> {
  const supabase = supabaseAdmin();
  const { data, error } = await supabase
    .from("chapters")
    .select("id")
    .eq("book_id", bookId)
    .order("order", { ascending: true });
  if (error || !data) return;
  for (let i = 0; i < data.length; i++) {
    await supabase.from("chapters").update({ order: i + 1 }).eq("id", data[i].id);
  }
}

async function refreshBookWordCount(bookId: string): Promise<void> {
  const supabase = supabaseAdmin();
  const { data } = await supabase
    .from("chapters")
    .select("word_count")
    .eq("book_id", bookId);
  const total = (data ?? []).reduce((sum, c: any) => sum + (c.word_count || 0), 0);
  await supabase.from("books").update({ total_word_count: total }).eq("id", bookId);
}
```

### 4.4 TXT 章节解析器

#### `src/lib/import/parser.ts`

```ts
export type ParsedChapter = {
  order: number;
  title: string;
  content: string;
};

/**
 * 从 TXT 文本中解析章节列表
 * 识别 "第X章"、"第X回"、"Chapter X" 等常见格式
 */
export function parseTxtToChapters(text: string): ParsedChapter[] {
  const lines = text.split(/\r?\n/);

  // 章节起始行的正则
  const chapterRegex =
    /^\s*(第\s*[一二三四五六七八九十百千万\d]+\s*[章回卷集部篇])\s*(.*)$/i;

  const chapters: ParsedChapter[] = [];
  let current: ParsedChapter | null = null;
  let contentLines: string[] = [];

  for (const line of lines) {
    const match = line.match(chapterRegex);
    if (match) {
      // 保存上一章
      if (current) {
        current.content = contentLines.join("\n").trim();
        chapters.push(current);
      }
      const title = (match[1] + (match[2] ? " " + match[2] : "")).trim();
      current = {
        order: chapters.length + 1,
        title,
        content: "",
      };
      contentLines = [];
    } else if (current) {
      contentLines.push(line);
    }
  }

  // 保存最后一章
  if (current) {
    current.content = contentLines.join("\n").trim();
    chapters.push(current);
  }

  // 如果没有识别到章节，当作单章全文
  if (chapters.length === 0 && text.trim()) {
    chapters.push({
      order: 1,
      title: "正文",
      content: text.trim(),
    });
  }

  // 过滤空内容
  return chapters.filter((c) => c.content.length > 0);
}
```

### 4.5 批量导入 API

#### `src/app/api/admin/import/route.ts`

```ts
import { NextResponse } from "next/server";
import { writeFile, mkdir } from "fs/promises";
import path from "path";
import { parseTxtToChapters } from "@/lib/import/parser";
import { supabaseAdmin } from "@/lib/supabase/admin";
import { verifyToken } from "@/lib/auth";
import { cookies } from "next/headers";

export const maxDuration = 60;

export async function POST(
  request: Request,
  { params }: { params: Promise<{ bookId: string }> }
) {
  const { bookId } = await params;

  // 鉴权
  const cookieStore = await cookies();
  const token = cookieStore.get("admin_token")?.value;
  if (!token || !verifyToken(token)) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }

  const form = await request.formData();
  const file = form.get("file") as File | null;
  if (!file) {
    return NextResponse.json({ error: "缺少文件" }, { status: 400 });
  }

  const text = await file.text();
  const chapters = parseTxtToChapters(text);

  if (chapters.length === 0) {
    return NextResponse.json({ error: "未能解析出任何章节" }, { status: 400 });
  }

  const supabase = supabaseAdmin();

  // 查询当前最大 order
  const { data: maxRow } = await supabase
    .from("chapters")
    .select("order")
    .eq("book_id", bookId)
    .order("order", { ascending: false })
    .limit(1)
    .maybeSingle();
  let baseOrder = (maxRow?.order ?? 0) + 1;

  // 批量插入
  for (const ch of chapters) {
    const wc = ch.content.replace(/\s/g, "").length;
    const { data, error } = await supabase
      .from("chapters")
      .insert({
        book_id: bookId,
        order: baseOrder++,
        title: ch.title,
        word_count: wc,
      })
      .select()
      .single();
    if (error) throw error;
    await supabase
      .from("chapter_contents")
      .insert({ chapter_id: data!.id, content: ch.content });
  }

  // 更新书籍总字数
  const { data: allChapters } = await supabase
    .from("chapters")
    .select("word_count")
    .eq("book_id", bookId);
  const total = (allChapters ?? []).reduce((sum, c: any) => sum + (c.word_count || 0), 0);
  await supabase.from("books").update({ total_word_count: total }).eq("id", bookId);

  return NextResponse.json({
    ok: true,
    imported: chapters.length,
  });
}
```

## M7：性能优化与字号调节

> 阶段目标：解决目录页 / 阅读页加载慢（5 秒+）的问题，并新增正文字号调节功能。
> 优化后目录页、阅读页首屏均能秒开，章节列表按需懒加载。

### 7.1 问题根因分析

通过浏览器 DevTools 网络瀑布图定位到三类瓶颈：

1. **Supabase 客户端反复创建**：`supabaseServer()` 每次调用都 `createClient()`，连接无法复用，TLS 握手开销叠加。
2. **目录页一次性拉取全部章节**：一本书 1400+ 章，服务端一次性 `select * from chapters where book_id=?` 把所有章节塞进 RSC payload，首屏字节量大、序列化慢。
3. **阅读页侧边栏预加载全部章节**：进入阅读页时为了渲染侧边栏目录，同样拉取了整本书的章节数据，拖慢首屏。

### 7.2 优化方案：按需懒加载 + 客户端单例

核心思路：**首屏只拉必需数据，次要数据交互时再取**。

| 页面 | 优化前 | 优化后 |
|------|--------|--------|
| 目录页 | 一次性加载卷 + 全部章节 | 只加载卷列表，展开某卷时 fetch 该卷章节 |
| 阅读页 | 加载章节 + 全部卷 + 全部章节 | 只加载当前章节 + 当前卷章节（用于翻页），侧边栏打开时再 fetch 全部目录 |
| Supabase | 每次 `createClient` | 模块级单例缓存 |

### 7.3 Supabase 客户端单例化

重写 `src/lib/supabase/server.ts`：

```ts
import { createClient } from "@supabase/supabase-js";

let cachedClient: ReturnType<typeof createClient> | null = null;

export function supabaseServer() {
  if (cachedClient) return cachedClient;
  cachedClient = createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
  );
  return cachedClient;
}
```

> 说明：模块级变量在 Node 运行时内复用，避免每次查询重新构造客户端、重新走一遍初始化流程。ISR 缓存命中时更不会触发任何查询。

```tsx
    volumeTitle={volume.title}
    volumeNumber={index + 1}
    chapters={[]}           // 初始为空，展开时懒加载
    defaultOpen={index === 0}
  />
))}
```

**客户端** `src/components/site/VolumeSection.tsx`：展开时调用 `/api/volumes/[volumeId]/chapters` 拉取该卷章节，带 loading 转圈，加载结果缓存在组件 state，避免重复请求。

```tsx
const loadChapters = async () => {
  setLoading(true);
  try {
    const res = await fetch(`/api/volumes/${volumeId}/chapters`);
    if (res.ok) {
      const data = await res.json();
      setChapters(data.chapters);
      setLoaded(true);
    }
  } finally {
    setLoading(false);
  }
};

useEffect(() => {
  if (open && !loaded && !loading) loadChapters();
}, [open, loaded, loading]);
```

### 7.6 新增 API 路由

新建 `src/app/api/books/[bookId]/volumes/route.ts`，返回某本书的全部卷 + 章节标题（轻量字段，不含正文）：

```ts
export async function GET(request: Request, { params }: { params: Promise<{ bookId: string }> }) {
  const { bookId } = await params;
  const { volumes, chaptersMap } = await fetchVolumesWithAllChapters(bookId);

  const result = volumes.map((v) => ({
    id: v.id,
    title: v.title,
    order: v.order,
    chapters: (chaptersMap.get(v.id) || []).map((ch) => ({
      id: ch.id,
      title: ch.title,
      order: ch.order,
      word_count: ch.word_count,
    })),
  }));

  return NextResponse.json(
    { volumes: result },
    { headers: { "Cache-Control": "public, s-maxage=1800, stale-while-revalidate=60" } }
  );
}

export const revalidate = 1800;
```

> 原有的 `/api/volumes/[volumeId]/chapters` 路由保留，供目录页单卷懒加载使用。两个 API 各司其职：单卷接口字段更少、更快；全卷接口用于阅读页侧边栏一次性铺开。

### 7.8 数据库索引优化

**必须手动在 Supabase SQL Editor 执行** `scripts/optimize.sql`：

```sql
-- sections 表：按 chapter_id 查正文
CREATE INDEX IF NOT EXISTS idx_sections_chapter_id ON sections(chapter_id);

-- chapters 表：按 book_id 取章节并排序
CREATE INDEX IF NOT EXISTS idx_chapters_book_order ON chapters(book_id, "order");

-- chapters 表：按 volume_id 取某卷章节
CREATE INDEX IF NOT EXISTS idx_chapters_volume_id ON chapters(volume_id);

-- volumes 表：按 book_id 取卷并排序
CREATE INDEX IF NOT EXISTS idx_volumes_book_order ON volumes(book_id, "order");
```

> 这一步对查询速度影响最大，未建索引时 1400+ 行的 chapters 表全表扫描是主要耗时来源。
