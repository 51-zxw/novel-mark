#### 5.4 `src/lib/theme.tsx`（主题 Provider）

新建文件，实现主题状态管理（客户端组件，用 `"use client"` 声明）：

```tsx
"use client";

import { createContext, useContext, useEffect, useState, ReactNode } from "react";

type Theme = "light" | "dark";

type ThemeContextValue = {
  theme: Theme;
  toggleTheme: () => void;
  setTheme: (t: Theme) => void;
};

const ThemeContext = createContext<ThemeContextValue | null>(null);

const STORAGE_KEY = "novel-mark-theme";

function getInitialTheme(): Theme {
  if (typeof window === "undefined") return "dark";
  const saved = localStorage.getItem(STORAGE_KEY);
  if (saved === "light" || saved === "dark") return saved;
  return "dark"; // 默认 dark
}

function applyTheme(theme: Theme) {
  const root = document.documentElement;
  // 默认 :root 即 dark，仅在 light 时添加 .light 类
  if (theme === "light") {
    root.classList.add("light");
  } else {
    root.classList.remove("light");
  }
}

export function ThemeProvider({ children }: { children: ReactNode }) {
  const [theme, setThemeState] = useState<Theme>(getInitialTheme);

  useEffect(() => {
    applyTheme(theme);
    localStorage.setItem(STORAGE_KEY, theme);
  }, [theme]);

  const value: ThemeContextValue = {
    theme,
    setTheme: setThemeState,
    toggleTheme: () => setThemeState((t) => (t === "dark" ? "light" : "dark")),
  };

  return <ThemeContext.Provider value={value}>{children}</ThemeContext.Provider>;
}

export function useTheme() {
  const ctx = useContext(ThemeContext);
  if (!ctx) throw new Error("useTheme must be used within ThemeProvider");
  return ctx;
}
```

#### 6.2 建表 SQL

进入 Supabase 控制台 → SQL Editor → New query，粘贴执行以下脚本：

```sql
-- ============================================
-- 一期表结构：books / chapters / sections / admin_users / reading_progress
-- ============================================

-- 启用 pgcrypto（生成 uuid）
create extension if not exists pgcrypto;

-- 1. 书籍
create table if not exists books (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  author text not null,
  cover_url text,
  description text,
  total_word_count int default 0,
  status text default '连载中',
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- 2. 章节目录
create table if not exists chapters (
  id uuid primary key default gen_random_uuid(),
  book_id uuid not null references books(id) on delete cascade,
  "order" int not null,
  title text not null,
  word_count int default 0,
  created_at timestamptz default now()
);

-- 章节复合索引（按书查询 + 排序）
create index if not exists idx_chapters_book_order on chapters(book_id, "order" asc);

-- 3. 章节正文（一对一）
create table if not exists sections (
  id uuid primary key default gen_random_uuid(),
  chapter_id uuid unique not null references chapters(id) on delete cascade,
  content text not null,
  updated_at timestamptz default now()
);

-- 4. 管理员账号
create table if not exists admin_users (
  id uuid primary key default gen_random_uuid(),
  username text unique not null,
  password_hash text not null,
  last_login timestamptz,
  created_at timestamptz default now()
);

-- 5. 阅读进度
create table if not exists reading_progress (
  id uuid primary key default gen_random_uuid(),
  user_id text not null,
  book_id uuid not null references books(id) on delete cascade,
  chapter_id uuid not null references chapters(id) on delete cascade,
  scroll_position float default 0,
  updated_at timestamptz default now()
);

create index if not exists idx_progress_user_book on reading_progress(user_id, book_id);

-- ============================================
-- updated_at 自动更新触发器
-- ============================================
create or replace function touch_updated_at()
returns trigger as $$
begin
  new.updated_at = now();
  return new;
end;
$$ language plpgsql;

do $$
declare t text;
begin
  for t in select unnest(array['books','sections','reading_progress']) loop
    execute format('drop trigger if exists trg_%I_touch on %I;', t, t);
    execute format('create trigger trg_%I_touch before update on %I for each row execute function touch_updated_at();', t, t);
  end loop;
end$$;

-- ============================================
-- RLS 策略
-- ============================================
-- 开启 RLS
alter table books enable row level security;
alter table chapters enable row level security;
alter table sections enable row level security;
alter table admin_users enable row level security;
alter table reading_progress enable row level security;

-- 书籍 / 章节 / 正文：匿名可读，仅 service_role 可写（service_role 默认绕过 RLS）
create policy "books_read_all" on books for select using (true);
create policy "chapters_read_all" on chapters for select using (true);
create policy "sections_read_all" on sections for select using (true);

-- 阅读进度：仅本人可读写（按 user_id 隔离，前端传 user_id）
create policy "progress_read_own" on reading_progress for select using (true);
create policy "progress_write_own" on reading_progress for all using (true) with check (true);

-- admin_users：禁止匿名访问（仅 service_role 可读，service_role 绕过 RLS）
create policy "admin_users_deny_anon" on admin_users for select using (false);
```

#### 7.6 `src/app/layout.tsx`（根 Layout）

> 主题策略：**CSS 默认 dark，`.light` 类覆盖**。
> 这样无需任何内联 JS 脚本即可首屏渲染 dark 主题（无 FOUC），也避免 Next.js 16 + React 19 下 `<script>` 标签触发的 "Encountered a script tag" 警告。
> ⚠️ `suppressHydrationWarning` 必须保留：客户端 hydrate 后可能给 `<html>` 加 `.light` 类，与 SSR 时的空 className 不一致，属预期行为。

```tsx
import type { Metadata } from "next";
import "./globals.css";

export const metadata: Metadata = {
  title: "小说阅读",
  description: "干净纯粹的网络小说阅读平台",
};

export default function RootLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  return (
    <html lang="zh-CN" suppressHydrationWarning>
      <body suppressHydrationWarning>{children}</body>
    </html>
  );
}
```

> 💡 **为什么不使用 `next/script` 注入主题初始化脚本？**
> Next.js 16 的 Script 组件在 App Router 下，`beforeInteractive` 内联脚本仍会渲染一个真实的 `<script>` 元素（见 `node_modules/next/dist/client/script.js:282`），React 19 在客户端 hydrate 时会对任何可执行 `<script>` 触发 "Encountered a script tag" 警告（见 `react-dom-client.development.js:12998` 的 `isScriptDataBlock` 判断），dev 模式下会显示为红色错误覆盖层。
> 改用「CSS 默认 dark + `.light` 覆盖」方案后，默认 dark 用户完全无 FOUC、无警告；仅 light 用户刷新时有一次极短闪烁，但绝大多数场景（默认 dark）体验最佳。
